const canvas = document.getElementById('canvas'); const ctx = canvas.getContext('2d'); const btn = document.getElementById('spin-btn'); const chanceDisplay = document.getElementById('chance-count'); const unlockArea = document.getElementById('unlock-area'); const BACKEND_URL = 'https://jn-lottery-api-arzjhjubux.cn-hangzhou.fcapp.run'; const prizeConfig = [ { name: "特等奖", color: "#ee5253", startDeg: 0, endDeg: 10 }, { name: "一等奖", color: "#f39c12", startDeg: 10, endDeg: 65 }, { name: "二等奖", color: "#aadd44", startDeg: 65, endDeg: 180 }, { name: "三等奖", color: "#1abc9c", startDeg: 180, endDeg: 360 } ]; let currentRotation = 0; // 从本地存储读取剩余次数，如果没有则默认为 1 次（第一次免费） let savedChances = localStorage.getItem('jnu_remaining_chances'); let chances = savedChances !== null ? parseInt(savedChances) : 1; // 初始化页面状态 function init() { chanceDisplay.innerText = chances; unlockArea.style.display = 'block'; // 永久显示兑换入口 if (chances <= 0) { btn.disabled = true; btn.innerText = "次数已用完，请兑换"; } else { btn.disabled = false; btn.innerText = "开始抽奖"; } } // 增加次数逻辑 (无限充值) function activateMoreChance() { const code = prompt("请找工作人员输入兑换码以增加抽奖次数："); if (code === "JNU888") { chances += 1; // 每输入一次加 1 次 localStorage.setItem('jnu_remaining_chances', chances); alert("兑换成功！当前剩余次数：" + chances); init(); // 刷新按钮状态 } else { alert("兑换码无效。"); } } async function handleSpin() { if (chances <= 0) return; btn.disabled = true; btn.innerText = "正在请求奖池..."; try { const response = await fetch(BACKEND_URL, { method: 'POST' }); const data = await response.json(); if (data.error) { alert("奖品已全部发完，下次早点来哦！"); btn.innerText = "活动结束"; return; } // 扣减次数 chances -= 1; localStorage.setItem('jnu_remaining_chances', chances); chanceDisplay.innerText = chances; const prize = prizeConfig.find(p => p.name === data.prize); const midDeg = (prize.startDeg + prize.endDeg) / 2; const targetCssDeg = (360 - midDeg) - 90; const finalRotation = currentRotation + 3600 + targetCssDeg; currentRotation = finalRotation; canvas.style.transform = `rotate(${finalRotation}deg)`; setTimeout(() => { alert(`🎉 恭喜！你抽中了【${data.prize}】\n当前剩余次数：${chances}`); init(); // 抽完后根据剩余次数决定按钮状态 }, 6500); } catch (e) { alert("网络繁忙，请稍后再试"); btn.disabled = false; btn.innerText = "重试"; } } function draw() { ctx.textAlign = "center"; ctx.font = "bold 17px sans-serif"; prizeConfig.forEach(p => { const s = (p.startDeg * Math.PI)/180, e = (p.endDeg * Math.PI)/180; ctx.beginPath(); ctx.fillStyle = p.color; ctx.moveTo(200, 200); ctx.arc(200, 200, 200, s, e); ctx.fill(); ctx.save(); ctx.fillStyle = "#fff"; ctx.translate(200, 200); ctx.rotate((s+e)/2); ctx.fillText(p.name, 130, 8); ctx.restore(); }); } draw(); init();
